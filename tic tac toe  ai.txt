board = [" " for _ in range(9)]


def display_board():
    print()
    print(board[0], "|", board[1], "|", board[2])
    print("--+---+--")
    print(board[3], "|", board[4], "|", board[5])
    print("--+---+--")
    print(board[6], "|", board[7], "|", board[8])
    print()


def check_winner():
    winning_positions = [
        (0, 1, 2),
        (3, 4, 5),
        (6, 7, 8),
        (0, 3, 6),
        (1, 4, 7),
        (2, 5, 8),
        (0, 4, 8),
        (2, 4, 6)
    ]

    for a, b, c in winning_positions:
        if board[a] == board[b] == board[c] and board[a] != " ":
            return board[a]

    return None


def board_full():
    return " " not in board


def minimax(is_maximizing):
    winner = check_winner()

    if winner == "O":
        return 1

    if winner == "X":
        return -1

    if board_full():
        return 0

    if is_maximizing:
        best_score = -float("inf")

        for i in range(9):
            if board[i] == " ":
                board[i] = "O"
                score = minimax(False)
                board[i] = " "
                best_score = max(best_score, score)

        return best_score

    else:
        best_score = float("inf")

        for i in range(9):
            if board[i] == " ":
                board[i] = "X"
                score = minimax(True)
                board[i] = " "
                best_score = min(best_score, score)

        return best_score


def computer_move():
    best_score = -float("inf")
    best_move = None

    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(False)
            board[i] = " "

            if score > best_score:
                best_score = score
                best_move = i

    board[best_move] = "O"


# Main game
print("TIC-TAC-TOE")
print("You are X")
print("Computer is O")

while True:

    display_board()

    # Human move
    while True:
        try:
            row = int(input("Enter row (1-3): "))
            col = int(input("Enter column (1-3): "))

            position = (row - 1) * 3 + (col - 1)

            if row < 1 or row > 3 or col < 1 or col > 3:
                print("Invalid position. Try again.")
            elif board[position] != " ":
                print("Position already occupied. Try again.")
            else:
                board[position] = "X"
                break

        except ValueError:
            print("Please enter numbers only.")

    # Check X win
    if check_winner() == "X":
        display_board()
        print("X wins!")
        break

    # Check draw
    if board_full():
        display_board()
        print("DRAW!")
        break

    # Computer move
    print("Computer is thinking...")
    computer_move()

    # Check O win
    if check_winner() == "O":
        display_board()
        print("O wins!")
        break

    # Check draw
    if board_full():
        display_board()
        print("DRAW!")
        break
